--  HMC (Hostel Management System) - DDL Script


DROP SCHEMA IF EXISTS hmc CASCADE;
CREATE SCHEMA hmc;
SET search_path = hmc;




--  ADMIN


CREATE TABLE Admin (
    admin_id    INT             PRIMARY KEY,
    name        VARCHAR(100)    NOT NULL,
    email       VARCHAR(150)    NOT NULL UNIQUE
);




--  STAFF (supertype)


CREATE TABLE Staff (
    staff_id        INTEGER         PRIMARY KEY,
    name            VARCHAR(100)    NOT NULL,
    contact_no      VARCHAR(15)     NOT NULL,
    working_hours   VARCHAR(30),
    working_days    VARCHAR(20),
    salary          NUMERIC(10,2),
    role            VARCHAR(50)     NOT NULL,
    managed_by      INTEGER         REFERENCES Admin(admin_id)
);




--  STAFF SUBTYPES


CREATE TABLE Cleaning_Staff (
    staff_id        INTEGER         PRIMARY KEY REFERENCES Staff(staff_id) ON DELETE CASCADE,
    working_area    VARCHAR(100)
);


CREATE TABLE Plumber (
    staff_id        INTEGER         PRIMARY KEY REFERENCES Staff(staff_id) ON DELETE CASCADE
);


CREATE TABLE Carpenter (
    staff_id        INTEGER         PRIMARY KEY REFERENCES Staff(staff_id) ON DELETE CASCADE
);


CREATE TABLE Electrician (
    staff_id        INTEGER         PRIMARY KEY REFERENCES Staff(staff_id) ON DELETE CASCADE
);




CREATE TABLE Warden (
    staff_id        INTEGER         PRIMARY KEY REFERENCES Staff(staff_id) ON DELETE CASCADE,
    assigned_wing   INTEGER         REFERENCES Wing(wing_id)
);




--  WING


CREATE TABLE Wing (
    wing_id     INTEGER         PRIMARY KEY,
    wing_name   VARCHAR(50)     NOT NULL UNIQUE,
    type        VARCHAR(50),
    floors      INTEGER
);




--  ROOMS


CREATE TABLE Rooms (
    wing_id         INTEGER,
    room_no         INTEGER,
    capacity        INTEGER         NOT NULL,
    availability    BOOLEAN         DEFAULT TRUE,
    PRIMARY KEY (wing_id, room_no),
    FOREIGN KEY (wing_id) REFERENCES Wing(wing_id)
);




--  STUDENT


CREATE TABLE Student (
    student_id          BIGINT          PRIMARY KEY,
    name                VARCHAR(100)    NOT NULL,
    dob                 DATE,
    contact             VARCHAR(15),
    email               VARCHAR(150)    UNIQUE,
    branch              VARCHAR(100),
    emergency_contact   VARCHAR(15),
    father_contact      VARCHAR(15),
    father_name         VARCHAR(100),
    year                INTEGER,
    health_details      TEXT,
    status              VARCHAR(30)     DEFAULT 'Active',
    program             VARCHAR(100),
    resident_wing       INTEGER,
    resident_room       INTEGER,
    hmc_flag            BOOLEAN         DEFAULT FALSE,
    gender              VARCHAR(10),
    FOREIGN KEY (resident_wing, resident_room) REFERENCES Rooms(wing_id, room_no)
);




--  VISITOR


CREATE TABLE Visitor (
    visitor_id      INTEGER          PRIMARY KEY,
    name            VARCHAR(100)    NOT NULL,
    contact_no      VARCHAR(15),
    aadhar_no       VARCHAR(12)     UNIQUE,
    stay_duration   INTEGER,
    check_in        DATE,
    check_out       DATE,
    request_status  VARCHAR(30)     DEFAULT 'Pending',
    approved_by     INTEGER         REFERENCES Admin(admin_id),
    allocated_wing INT,
    allocated_room INT,
    FOREIGN KEY (allocated_wing, allocated_room) REFERENCES Rooms(wing_id, room_no)
);




--  SUGGESTIONS


CREATE TABLE Suggestions (
    suggestion_id   INTEGER         PRIMARY KEY,
    description     TEXT            NOT NULL,
    visitor_id      INTEGER         NOT NULL REFERENCES Visitor(visitor_id)
);




--  PAYMENT (base transaction)


CREATE TABLE Payment (
    transaction_id  BIGINT          PRIMARY KEY,
    method          VARCHAR(50)     NOT NULL,
    date            DATE            NOT NULL
);




--  STUDENT_PAYMENT (specialization of Payment)


CREATE TABLE Student_Payment (
    transaction_id  BIGINT         PRIMARY KEY REFERENCES Payment(transaction_id) ON DELETE CASCADE,
    student_id      BIGINT         NOT NULL REFERENCES Student(student_id),
    amount          NUMERIC(10,2)   NOT NULL
);




--  VISITOR_PAYMENT (specialization of Payment)


CREATE TABLE Visitor_Payment (
    transaction_id  BIGINT         PRIMARY KEY REFERENCES Payment(transaction_id) ON DELETE CASCADE,
    visitor_id      BIGINT         NOT NULL REFERENCES Visitor(visitor_id),
    amount          NUMERIC(10,2)   NOT NULL
);




--  COMPLAINT


CREATE TABLE Complaint (
    comp_id         INTEGER          PRIMARY KEY,
    type            VARCHAR(50),
    status          VARCHAR(30)     DEFAULT 'Open',
    description     TEXT,
    reporting_date  DATE            NOT NULL,
    assigned_by     INTEGER         REFERENCES Staff(staff_id),
    student_id      BIGINT         NOT NULL REFERENCES Student(student_id),
    solved_by       INTEGER         REFERENCES Staff(staff_id)
);




--  PARCEL


CREATE TABLE Parcel (
    parcel_id       INTEGER          PRIMARY KEY,
    arrival_date    DATE            NOT NULL,
    timestamp       TIMESTAMP       DEFAULT NOW(),
    status          VARCHAR(30)     DEFAULT 'Pending',
    received_date   DATE,
    student_id      BIGINT         NOT NULL REFERENCES Student(student_id),
    handover_by     INTEGER         REFERENCES Staff(staff_id)
);




--  OVERNIGHT_LEAVE


CREATE TABLE Overnight_Leave (
    leave_id            INTEGER          PRIMARY KEY,
    request_date        DATE            NOT NULL,
    from_date           DATE            NOT NULL,
    to_date             DATE            NOT NULL,
    return_timestamp    TIMESTAMP,
    status              VARCHAR(30)     DEFAULT 'Pending',
    student_id          BIGINT         NOT NULL REFERENCES Student(student_id),
    approved_by         INTEGER         REFERENCES Staff(staff_id)
);




--  ATTENDANCE


CREATE TABLE Attendance (
    student_id  BIGINT         NOT NULL REFERENCES Student(student_id),
    date        DATE            NOT NULL,
    status      VARCHAR(20)     DEFAULT 'Present',
    PRIMARY KEY (student_id, date)
);




--  CONSTRAINTS: disjoint payment subtypes
--  A transaction_id cannot appear in BOTH Student_Payment
--  and Visitor_Payment simultaneously.


-- Enforced at application level; optionally via trigger:


CREATE OR REPLACE FUNCTION hmc.check_payment_disjoint()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_TABLE_NAME = 'student_payment' THEN
        IF EXISTS (SELECT 1 FROM hmc.Visitor_Payment WHERE transaction_id = NEW.transaction_id) THEN
            RAISE EXCEPTION 'Transaction % already exists in Visitor_Payment', NEW.transaction_id;
        END IF;
    ELSIF TG_TABLE_NAME = 'visitor_payment' THEN
        IF EXISTS (SELECT 1 FROM hmc.Student_Payment WHERE transaction_id = NEW.transaction_id) THEN
            RAISE EXCEPTION 'Transaction % already exists in Student_Payment', NEW.transaction_id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER trg_student_payment_disjoint
    BEFORE INSERT ON Student_Payment
    FOR EACH ROW EXECUTE FUNCTION hmc.check_payment_disjoint();


CREATE TRIGGER trg_visitor_payment_disjoint
    BEFORE INSERT ON Visitor_Payment
    FOR EACH ROW EXECUTE FUNCTION hmc.check_payment_disjoint();
